# ==================== CELL 1: Imports - Assignment: Time Series Prediction using RNN (Google Stock) ====================
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers
from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import mean_squared_error, mean_absolute_error
import warnings
warnings.filterwarnings('ignore')

np.random.seed(42)
tf.random.set_seed(42)


# ==================== CELL 2: Load data (put GOOGL.csv with Date and Close columns next to the notebook) ====================
df = pd.read_csv('GOOGL.csv')
df['Date'] = pd.to_datetime(df['Date'])
df = df.sort_values('Date').set_index('Date')
df.head()


# ==================== CELL 3: Inspect ====================
print("Shape:", df.shape)
df.info()
df.isnull().sum()


# ==================== CELL 4: Visualise Close price history ====================
plt.figure(figsize=(14, 6))
plt.plot(df['Close'])
plt.title('Google Stock Price History')
plt.xlabel('Date'); plt.ylabel('Close Price USD ($)')
plt.show()


# ==================== CELL 5: Keep only the Close column ====================
data = df[['Close']]
dataset = data.values
training_data_len = int(np.ceil(0.8 * len(dataset)))
print("Total rows:", len(dataset), " Train rows:", training_data_len)


# ==================== CELL 6: Scale to 0-1 (fit on TRAIN part only) ====================
scaler = MinMaxScaler(feature_range=(0, 1))
scaler.fit(dataset[:training_data_len])
scaled_data = scaler.transform(dataset)


# ==================== CELL 7: Create training sequences (window = 60 days) ====================
time_steps = 60
train_data = scaled_data[:training_data_len]

x_train, y_train = [], []
for i in range(time_steps, len(train_data)):
    x_train.append(train_data[i - time_steps:i, 0])
    y_train.append(train_data[i, 0])

x_train, y_train = np.array(x_train), np.array(y_train)
x_train = x_train.reshape(x_train.shape[0], x_train.shape[1], 1)
print("x_train:", x_train.shape, " y_train:", y_train.shape)


# ==================== CELL 8: Build the RNN (stacked LSTM) ====================
model = keras.Sequential([
    layers.Input(shape=(time_steps, 1)),
    layers.LSTM(50, return_sequences=True),
    layers.LSTM(50, return_sequences=False),
    layers.Dense(25),
    layers.Dense(1)
])
model.summary()


# ==================== CELL 9: Compile ====================
model.compile(optimizer='adam', loss='mean_squared_error')


# ==================== CELL 10: Train ====================
history = model.fit(x_train, y_train,
                    batch_size=32,
                    epochs=20,
                    validation_split=0.1)


# ==================== CELL 11: Plot training loss ====================
plt.plot(history.history['loss'], label='Train Loss')
plt.plot(history.history['val_loss'], label='Validation Loss')
plt.xlabel('Epoch'); plt.ylabel('MSE'); plt.title('LSTM Loss'); plt.legend()
plt.show()


# ==================== CELL 12: Create test sequences ====================
test_data = scaled_data[training_data_len - time_steps:]
y_test = dataset[training_data_len:]

x_test = []
for i in range(time_steps, len(test_data)):
    x_test.append(test_data[i - time_steps:i, 0])

x_test = np.array(x_test)
x_test = x_test.reshape(x_test.shape[0], x_test.shape[1], 1)
print("x_test:", x_test.shape)


# ==================== CELL 13: Predict and convert back to dollars ====================
predictions = model.predict(x_test)
predictions = scaler.inverse_transform(predictions)


# ==================== CELL 14: Evaluate ====================
rmse = np.sqrt(mean_squared_error(y_test, predictions))
mae = mean_absolute_error(y_test, predictions)
print("RMSE:", rmse)
print("MAE :", mae)


# ==================== CELL 15: Plot actual vs predicted ====================
train = data[:training_data_len]
valid = data[training_data_len:].copy()
valid['Predictions'] = predictions

plt.figure(figsize=(14, 6))
plt.title('Google Stock Price Prediction (LSTM)')
plt.xlabel('Date'); plt.ylabel('Close Price USD ($)')
plt.plot(train['Close'], label='Train')
plt.plot(valid['Close'], label='Actual')
plt.plot(valid['Predictions'], label='Predicted')
plt.legend(loc='lower right')
plt.show()


# ==================== CELL 16: Show last few actual vs predicted ====================
valid.tail(10)
