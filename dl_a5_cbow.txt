# ==================== CELL 1: Assignment 5: CBOW Model - Imports ====================
import re
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers
from sklearn.decomposition import PCA

np.random.seed(42)
tf.random.set_seed(42)


# ==================== CELL 2: a. DATA PREPARATION - the text ====================
data = """Deep learning (also known as deep structured learning) is part of a broader family of machine learning methods based on artificial neural networks with representation learning. Learning can be supervised, semi-supervised or unsupervised. Deep-learning architectures such as deep neural networks, deep belief networks, deep reinforcement learning, recurrent neural networks, convolutional neural networks and Transformers have been applied to fields including computer vision, speech recognition, natural language processing, machine translation, bioinformatics, drug design, medical image analysis, climate science, material inspection and board game programs, where they have produced results comparable to and in some cases surpassing human expert performance."""

sentences = data.split('.')
print(len(sentences), "sentences")


# ==================== CELL 3: Clean the sentences ====================
clean_sent = []
for sentence in sentences:
    if sentence.strip() == "":
        continue
    sentence = re.sub('[^A-Za-z0-9]+', ' ', sentence)
    sentence = re.sub(r'(?:^| )\w (?:$| )', ' ', sentence)
    sentence = sentence.strip().lower()
    clean_sent.append(sentence)

for s in clean_sent:
    print(s, "\n")


# ==================== CELL 4: Build vocabulary (word <-> index) ====================
all_words = []
for s in clean_sent:
    for w in s.split():
        if w not in all_words:
            all_words.append(w)

word_to_index = {w: i + 1 for i, w in enumerate(all_words)}
index_to_word = {i: w for w, i in word_to_index.items()}
vocab_size = len(word_to_index) + 1

print("Vocabulary size:", vocab_size)
print(word_to_index)


# ==================== CELL 5: Convert sentences to sequences of indices ====================
sequences = [[word_to_index[w] for w in s.split()] for s in clean_sent]
print(sequences)


# ==================== CELL 6: b. GENERATE TRAINING DATA (context -> target) ====================
context_size = 2
emb_size = 10

contexts = []
targets = []
for sequence in sequences:
    for i in range(context_size, len(sequence) - context_size):
        target = sequence[i]
        context = [sequence[i - 2], sequence[i - 1], sequence[i + 1], sequence[i + 2]]
        contexts.append(context)
        targets.append(target)

X = np.array(contexts)
Y = np.array(targets)
print("X:", X.shape, " Y:", Y.shape)


# ==================== CELL 7: Show a few training samples ====================
for i in range(5):
    context_words = [index_to_word[j] for j in contexts[i]]
    print(context_words, " -> ", index_to_word[targets[i]])


# ==================== CELL 8: c. TRAIN MODEL - build CBOW network ====================
model = keras.Sequential([
    layers.Input(shape=(2 * context_size,)),
    layers.Embedding(input_dim=vocab_size, output_dim=emb_size),
    layers.GlobalAveragePooling1D(),
    layers.Dense(256, activation='relu'),
    layers.Dense(512, activation='relu'),
    layers.Dense(vocab_size, activation='softmax')
])
model.summary()


# ==================== CELL 9: Compile and train ====================
model.compile(loss='sparse_categorical_crossentropy', optimizer='adam', metrics=['accuracy'])
history = model.fit(X, Y, epochs=80, verbose=1)


# ==================== CELL 10: Plot training loss / accuracy ====================
plt.figure(figsize=(12, 4))
plt.subplot(1, 2, 1)
plt.plot(history.history['loss']); plt.title('Loss'); plt.xlabel('Epoch')
plt.subplot(1, 2, 2)
plt.plot(history.history['accuracy']); plt.title('Accuracy'); plt.xlabel('Epoch')
plt.show()


# ==================== CELL 11: d. OUTPUT - test the model with context words ====================
test_sentences = [
    "known as structured learning",
    "transformers have applied to",
    "where they produced results",
    "cases surpassing expert performance"
]

for sent in test_sentences:
    test_words = sent.split(" ")
    x_test = np.array([[word_to_index.get(w, 0) for w in test_words]])
    pred = model.predict(x_test, verbose=0)
    pred_word = index_to_word.get(int(np.argmax(pred[0])), '?')
    print("Context:", test_words, "\n   => predicted center word:", pred_word, "\n")


# ==================== CELL 12: Word embeddings (learned vectors) ====================
embeddings = model.layers[0].get_weights()[0]
emb_df = pd.DataFrame(embeddings[1:], index=[index_to_word[i] for i in range(1, vocab_size)])
emb_df.head()


# ==================== CELL 13: Visualise embeddings in 2D using PCA ====================
reduced = PCA(n_components=2).fit_transform(embeddings[1:])

plt.figure(figsize=(12, 9))
plt.scatter(reduced[:, 0], reduced[:, 1])
for i in range(len(reduced)):
    plt.annotate(index_to_word[i + 1], (reduced[i, 0], reduced[i, 1]), fontsize=8)
plt.title('CBOW word embeddings (PCA 2D)')
plt.show()
