# ##########################################################
#  DL ASSIGNMENT 4: Anomaly Detection using Autoencoder
#  Dataset: ECG (5000 heartbeats, 140 points each). Label 1 = NORMAL, 0 = ANOMALY
#  Steps: a. Import libraries      b. Upload / access the dataset
#         c. Encoder -> latent representation
#         d. Decoder -> reconstruct the input
#         e. Compile with optimizer, loss and metrics
#  Idea: train the autoencoder ONLY on normal beats. Anomalies will
#  reconstruct badly (high error) -> flagged by a threshold.
# ##########################################################

# ==================== CELL 1: a. Imports ====================
import numpy as np
import pandas as pd
import tensorflow as tf
import matplotlib.pyplot as plt
from sklearn.metrics import accuracy_score, confusion_matrix, classification_report
from sklearn.preprocessing import MinMaxScaler
from sklearn.model_selection import train_test_split
from tensorflow.keras import Model, Sequential
from tensorflow.keras.layers import Dense, Dropout

np.random.seed(42)
tf.random.set_seed(42)


# ==================== CELL 2: b. Load the dataset ====================
path = 'https://storage.googleapis.com/download.tensorflow.org/data/ecg.csv'
# If you have it offline:  path = 'ecg.csv'
data = pd.read_csv(path, header=None)
data.head()


# ==================== CELL 3: Inspect ====================
print("Shape:", data.shape)
data.info()
print(data[140].value_counts())      # 1 = normal, 0 = anomaly


# ==================== CELL 4: Plot one normal and one anomalous ECG ====================
plt.figure(figsize=(12, 4))
plt.subplot(1, 2, 1)
plt.plot(data[data[140] == 1].iloc[0, :140]); plt.title('Normal ECG')
plt.subplot(1, 2, 2)
plt.plot(data[data[140] == 0].iloc[0, :140]); plt.title('Anomalous ECG')
plt.show()


# ==================== CELL 5: Split features / target, then train / test ====================
features = data.drop(140, axis=1)
target = data[140]

x_train, x_test, y_train, y_test = train_test_split(
    features, target, test_size=0.2, random_state=42, stratify=target
)

# Keep only NORMAL (label 1) beats for training the autoencoder
train_index = y_train[y_train == 1].index
train_data = x_train.loc[train_index]
print("Normal beats used for training:", train_data.shape)


# ==================== CELL 6: Scale to 0-1 (fit on training data only) ====================
min_max_scaler = MinMaxScaler(feature_range=(0, 1))
x_train_scaled = min_max_scaler.fit_transform(train_data.copy())
x_test_scaled = min_max_scaler.transform(x_test.copy())


# ==================== CELL 7: c + d. Autoencoder (Encoder + Decoder) ====================
class AutoEncoder(Model):
    def __init__(self, output_units, ldim=8):
        super().__init__()
        # c. Encoder: input -> latent representation (ldim = 8 values)
        self.encoder = Sequential([
            Dense(64, activation='relu'),
            Dropout(0.1),
            Dense(32, activation='relu'),
            Dropout(0.1),
            Dense(16, activation='relu'),
            Dropout(0.1),
            Dense(ldim, activation='relu')
        ])
        # d. Decoder: latent representation -> reconstructed input
        self.decoder = Sequential([
            Dense(16, activation='relu'),
            Dropout(0.1),
            Dense(32, activation='relu'),
            Dropout(0.1),
            Dense(64, activation='relu'),
            Dropout(0.1),
            Dense(output_units, activation='sigmoid')
        ])

    def call(self, inputs):
        encoded = self.encoder(inputs)
        decoded = self.decoder(encoded)
        return decoded


# ==================== CELL 8: e. Compile ====================
model = AutoEncoder(output_units=x_train_scaled.shape[1])
model.compile(loss='msle', metrics=['mse'], optimizer='adam')


# ==================== CELL 9: Train (input = target = same data) ====================
history = model.fit(
    x_train_scaled, x_train_scaled,
    epochs=20,
    batch_size=512,
    validation_data=(x_test_scaled, x_test_scaled),
    shuffle=True
)
model.summary()


# ==================== CELL 10: Plot loss ====================
plt.plot(history.history["loss"], label="Training Loss")
plt.plot(history.history["val_loss"], label="Validation Loss")
plt.xlabel('Epoch'); plt.ylabel('MSLE'); plt.legend()
plt.show()


# ==================== CELL 11: Threshold = mean + 1 std of reconstruction error on normal data ====================
def find_threshold(model, x_scaled):
    recons = model.predict(x_scaled)
    recons_error = tf.keras.losses.msle(recons, x_scaled).numpy()
    return np.mean(recons_error) + np.std(recons_error)

threshold = find_threshold(model, x_train_scaled)
print(f"Threshold: {threshold}")


# ==================== CELL 12: Predict anomalies on test data ====================
def get_predictions(model, x_scaled, threshold):
    predictions = model.predict(x_scaled)
    errors = tf.keras.losses.msle(predictions, x_scaled).numpy()
    # error above threshold -> anomaly (0), else normal (1)
    return np.where(errors > threshold, 0.0, 1.0), errors

preds, errors = get_predictions(model, x_test_scaled, threshold)


# ==================== CELL 13: Evaluate ====================
print("Accuracy Score:", accuracy_score(y_test, preds))
print("\nConfusion Matrix (rows=actual, cols=predicted; 0=anomaly, 1=normal):\n",
      confusion_matrix(y_test, preds))
print("\n", classification_report(y_test, preds, target_names=['Anomaly', 'Normal']))


# ==================== CELL 14: Reconstruction error distribution ====================
plt.hist(errors[y_test.values == 1], bins=50, alpha=0.6, label='Normal')
plt.hist(errors[y_test.values == 0], bins=50, alpha=0.6, label='Anomaly')
plt.axvline(threshold, color='r', linestyle='--', label='Threshold')
plt.xlabel('Reconstruction error'); plt.ylabel('Count'); plt.legend()
plt.show()
