# ##########################################################
#  DL ASSIGNMENT 2: Binary Classification using Deep Neural Network
#  Dataset: IMDB Movie Reviews (positive / negative)
# ##########################################################

# ==================== CELL 1: Imports ====================
import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers
from sklearn.metrics import classification_report, confusion_matrix, accuracy_score
import warnings
warnings.filterwarnings('ignore')


# ==================== CELL 2: Load data (top 10,000 most frequent words) ====================
(train_data, train_labels), (test_data, test_labels) = keras.datasets.imdb.load_data(num_words=10000)

print("Train reviews:", len(train_data), " Test reviews:", len(test_data))
print("First review (word indices):", train_data[0][:15])
print("First label (1=positive, 0=negative):", train_labels[0])


# ==================== CELL 3: Decode a review back to text ====================
word_index = keras.datasets.imdb.get_word_index()
reverse_word_index = {v: k for k, v in word_index.items()}
# indices 0,1,2 are reserved (padding, start, unknown) -> offset by 3
decoded = ' '.join([reverse_word_index.get(i - 3, '?') for i in train_data[0]])
print(decoded)


# ==================== CELL 4: Vectorize (multi-hot encode) ====================
def vectorize_sequences(sequences, dimension=10000):
    results = np.zeros((len(sequences), dimension))
    for i, seq in enumerate(sequences):
        results[i, seq] = 1.0
    return results

x_train = vectorize_sequences(train_data)
x_test = vectorize_sequences(test_data)
y_train = np.asarray(train_labels).astype('float32')
y_test = np.asarray(test_labels).astype('float32')

print("x_train:", x_train.shape, " x_test:", x_test.shape)


# ==================== CELL 5: Validation set (first 10,000 samples) ====================
x_val = x_train[:10000]
partial_x_train = x_train[10000:]
y_val = y_train[:10000]
partial_y_train = y_train[10000:]


# ==================== CELL 6: Build the Deep Neural Network ====================
model = keras.Sequential([
    layers.Input(shape=(10000,)),
    layers.Dense(16, activation='relu'),
    layers.Dense(16, activation='relu'),
    layers.Dense(1, activation='sigmoid')      # sigmoid -> probability of positive
])
model.summary()


# ==================== CELL 7: Compile ====================
model.compile(optimizer='rmsprop',
              loss='binary_crossentropy',
              metrics=['accuracy'])


# ==================== CELL 8: Train ====================
history = model.fit(partial_x_train, partial_y_train,
                    epochs=20,
                    batch_size=512,
                    validation_data=(x_val, y_val))


# ==================== CELL 9: Plot Loss ====================
h = history.history
epochs = range(1, len(h['loss']) + 1)

plt.plot(epochs, h['loss'], 'bo', label='Training Loss')
plt.plot(epochs, h['val_loss'], 'b', label='Validation Loss')
plt.title('Training and Validation Loss')
plt.xlabel('Epochs'); plt.ylabel('Loss'); plt.legend(); plt.show()


# ==================== CELL 10: Plot Accuracy ====================
plt.plot(epochs, h['accuracy'], 'ro', label='Training Accuracy')
plt.plot(epochs, h['val_accuracy'], 'r', label='Validation Accuracy')
plt.title('Training and Validation Accuracy')
plt.xlabel('Epochs'); plt.ylabel('Accuracy'); plt.legend(); plt.show()


# ==================== CELL 11: Retrain from scratch with fewer epochs (overfitting starts ~4) ====================
model = keras.Sequential([
    layers.Input(shape=(10000,)),
    layers.Dense(16, activation='relu'),
    layers.Dense(16, activation='relu'),
    layers.Dense(1, activation='sigmoid')
])
model.compile(optimizer='rmsprop', loss='binary_crossentropy', metrics=['accuracy'])
model.fit(x_train, y_train, epochs=4, batch_size=512, verbose=1)


# ==================== CELL 12: Evaluate on test data ====================
test_loss, test_acc = model.evaluate(x_test, y_test, verbose=0)
print("Test Loss    :", test_loss)
print("Test Accuracy:", test_acc)


# ==================== CELL 13: Predictions + classification report ====================
probs = model.predict(x_test)
y_pred = (probs > 0.5).astype('int32').flatten()

print("Accuracy:", accuracy_score(y_test, y_pred))
print(classification_report(y_test, y_pred, target_names=['Negative', 'Positive']))
print("Confusion Matrix:\n", confusion_matrix(y_test, y_pred))


# ==================== CELL 14: Check a few reviews ====================
for i in range(5):
    print(f"Review {i}: predicted = {'Positive' if y_pred[i] else 'Negative'} "
          f"(prob={probs[i][0]:.3f}), actual = {'Positive' if y_test[i] else 'Negative'}")
