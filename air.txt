# ==================== CELL 1: Imports ====================
import pandas as pd
import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt
from sklearn.preprocessing import LabelEncoder, MinMaxScaler


# ==================== CELL 2: Load data ====================
df = pd.read_csv('airquality_data.csv', encoding='cp1252', low_memory=False)
df.head()


# ==================== CELL 3: Inspect ====================
df.info()


# ##########################################################
#                   a. DATA CLEANING
# ##########################################################

# ==================== CELL 4: Reduce memory (float64 -> float32) ====================
num_cols = ['so2', 'no2', 'rspm', 'spm']
for col in num_cols:
    df[col] = df[col].astype('float32')
df.info()


# ==================== CELL 5: Remove duplicates ====================
print("Rows before:", len(df))
df = df.drop_duplicates()
print("Rows after :", len(df))


# ==================== CELL 6: Percentage of missing values ====================
percent_missing = df.isnull().sum() * 100 / len(df)
percent_missing.sort_values(ascending=False)


# ==================== CELL 7: Drop mostly-empty / not useful columns ====================
df = df.drop(['stn_code', 'agency', 'sampling_date',
              'location_monitoring_station', 'pm2_5'], axis=1)
df.columns


# ==================== CELL 8: Fill missing values (mean for numbers, mode for text) ====================
for col in df.columns:
    if pd.api.types.is_numeric_dtype(df[col]):
        df[col] = df[col].fillna(df[col].mean())
    else:
        df[col] = df[col].fillna(df[col].mode()[0])

df.isna().sum()


# ##########################################################
#                   b. ERROR CORRECTING
# ##########################################################

# ==================== CELL 9: Fix inconsistent category labels ====================
print("Before:\n", df['type'].value_counts(), "\n")

df['type'] = df['type'].str.strip()
df['type'] = df['type'].replace({
    'Industrial Areas': 'Industrial Area',
    'Industrial': 'Industrial Area',
    'Sensitive Areas': 'Sensitive Area',
    'Sensitive': 'Sensitive Area'
})
df['state'] = df['state'].str.strip()
df['location'] = df['location'].str.strip()

print("After:\n", df['type'].value_counts())


# ==================== CELL 10: Handle outliers with IQR capping ====================
def cap_outliers(column):
    Q1 = column.quantile(0.25)
    Q3 = column.quantile(0.75)
    IQR = Q3 - Q1
    lower = Q1 - 1.5 * IQR
    upper = Q3 + 1.5 * IQR
    return column.clip(lower, upper)

for col in num_cols:
    df[col] = cap_outliers(df[col])

df[num_cols].describe()


# ==================== CELL 11: Boxplots after outlier handling ====================
for col in num_cols:
    plt.figure(figsize=(8, 4))
    sns.boxplot(x=df[col])
    plt.title(f'{col} after capping outliers')
    plt.show()


# ##########################################################
#                   c. DATA INTEGRATION
# ##########################################################

# ==================== CELL 12: Merge state-level averages on common key 'state' ====================
state_avg = (df.groupby('state')[['so2', 'no2']]
               .mean()
               .add_suffix('_state_avg')
               .reset_index())

integrated_df = pd.merge(df, state_avg, on='state', how='left')
integrated_df.head()


# ##########################################################
#                   d. DATA TRANSFORMATION
# ##########################################################

# ==================== CELL 13: Convert date, extract year and month ====================
df['date'] = pd.to_datetime(df['date'], errors='coerce')
df = df.dropna(subset=['date'])
df['year'] = df['date'].dt.year
df['month'] = df['date'].dt.month
df.head()


# ==================== CELL 14: Label-encode only categorical columns ====================
cat_cols = ['state', 'location', 'type']
encoder = LabelEncoder()
for col in cat_cols:
    df[col] = encoder.fit_transform(df[col])
df.head()


# ==================== CELL 15: Min-Max normalize pollutant columns (0 to 1) ====================
scaler = MinMaxScaler()
df[num_cols] = scaler.fit_transform(df[num_cols])
df.head()
