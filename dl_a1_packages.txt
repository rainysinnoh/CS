# ==================== CELL 2: Same tiny computation in each framework ====================
# y = (a * b) + c   with a=2, b=3, c=4  -> expected 10

# ---- TensorFlow ----
try:
    import tensorflow as tf
    a, b, c = tf.constant(2.0), tf.constant(3.0), tf.constant(4.0)
    print("TensorFlow:", (a * b + c).numpy())
except Exception as e:
    print("TensorFlow error:", e)

# ---- Keras (tiny model: one Dense layer) ----
try:
    from tensorflow import keras
    import numpy as np
    m = keras.Sequential([keras.layers.Input(shape=(1,)), keras.layers.Dense(1)])
    m.compile(optimizer='sgd', loss='mse')
    print("Keras: model built, params =", m.count_params())
except Exception as e:
    print("Keras error:", e)

# ---- PyTorch ----
try:
    import torch
    a, b, c = torch.tensor(2.0), torch.tensor(3.0), torch.tensor(4.0)
    print("PyTorch:", (a * b + c).item())
except Exception as e:
    print("PyTorch error:", e)

# ---- Theano (symbolic) ----
try:
    import theano
    import theano.tensor as T
    x, y, z = T.dscalars('x', 'y', 'z')
    f = theano.function([x, y, z], x * y + z)
    print("Theano:", f(2, 3, 4))
except Exception as e:
    print("Theano not available:", type(e).__name__)

study = """

   Distinct features:
   - Open-source deep learning framework developed by Google.

   - Computational-graph based, which makes it efficient for deep learning.

   - Open-source deep learning framework developed by Facebook's AI Research lab (now Meta).

   Functionality:
   - Tensors with GPU acceleration, autograd for automatic differentiation.
   - Libraries for NLP (torchtext), computer vision (torchvision), audio (torchaudio).
