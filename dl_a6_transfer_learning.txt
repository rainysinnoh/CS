# ==================== CELL 1: Assignment 6: Transfer Learning with VGG16 (CIFAR-10) - Imports ====================
import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers
from tensorflow.keras.applications import VGG16
from tensorflow.keras.applications.vgg16 import preprocess_input

np.random.seed(42)
tf.random.set_seed(42)


# ==================== CELL 2: Load CIFAR-10 ====================
(x_train, y_train), (x_test, y_test) = keras.datasets.cifar10.load_data()

class_names = ['airplane', 'automobile', 'bird', 'cat', 'deer',
               'dog', 'frog', 'horse', 'ship', 'truck']

x_train, y_train = x_train[:10000], y_train[:10000]
x_test, y_test = x_test[:2000], y_test[:2000]
raw_test = x_test[:10].copy()

print(x_train.shape, y_train.shape, x_test.shape, y_test.shape)


# ==================== CELL 3: Preprocess for VGG16 + one-hot labels ====================
x_train = preprocess_input(x_train.astype('float32'))
x_test = preprocess_input(x_test.astype('float32'))

y_train_cat = keras.utils.to_categorical(y_train, 10)
y_test_cat = keras.utils.to_categorical(y_test, 10)


# ==================== CELL 4: a. Load pre-trained VGG16 (needs internet once for ImageNet weights) ====================
vgg_model = VGG16(weights='imagenet', include_top=False, input_shape=(32, 32, 3))
vgg_model.summary()


# ==================== CELL 5: b. Freeze all convolutional layers ====================
vgg_model.trainable = False

for layer in vgg_model.layers:
    print(f"{layer.name:20s} trainable = {layer.trainable}")


# ==================== CELL 6: c. Add custom classifier ====================
model = keras.Sequential([
    vgg_model,
    layers.Flatten(),
    layers.Dense(100, activation='relu'),
    layers.Dropout(0.2),
    layers.Dense(100, activation='relu'),
    layers.Dropout(0.2),
    layers.Dense(10, activation='softmax')
])
model.summary()


# ==================== CELL 7: Compile ====================
model.compile(optimizer=keras.optimizers.Adam(learning_rate=1e-3),
              loss='categorical_crossentropy',
              metrics=['accuracy'])


# ==================== CELL 8: d. Train ONLY the new classifier layers ====================
history1 = model.fit(x_train, y_train_cat,
                     batch_size=64,
                     epochs=5,
                     validation_data=(x_test, y_test_cat))


# ==================== CELL 9: e. Fine-tune - unfreeze the last 4 VGG16 layers ====================
vgg_model.trainable = True
for layer in vgg_model.layers[:-4]:
    layer.trainable = False

for layer in vgg_model.layers:
    print(f"{layer.name:20s} trainable = {layer.trainable}")

model.compile(optimizer=keras.optimizers.Adam(learning_rate=1e-5),
              loss='categorical_crossentropy',
              metrics=['accuracy'])

history2 = model.fit(x_train, y_train_cat,
                     batch_size=64,
                     epochs=5,
                     validation_data=(x_test, y_test_cat))


# ==================== CELL 10: Plot accuracy (before and after fine-tuning) ====================
acc = history1.history['val_accuracy'] + history2.history['val_accuracy']
plt.plot(acc, marker='o')
plt.axvline(len(history1.history['val_accuracy']) - 0.5, color='r', linestyle='--', label='Fine-tuning starts')
plt.xlabel('Epoch'); plt.ylabel('Validation accuracy'); plt.legend()
plt.title('Transfer learning')
plt.show()


# ==================== CELL 11: Evaluate ====================
loss, acc = model.evaluate(x_test, y_test_cat)
print("Test Loss    :", loss)
print("Test Accuracy:", acc)


# ==================== CELL 12: Predict a few images ====================
pred = model.predict(x_test[:10])
plt.figure(figsize=(14, 3))
for i in range(10):
    plt.subplot(1, 10, i + 1)
    plt.imshow(raw_test[i])
    ok = np.argmax(pred[i]) == y_test[i][0]
    plt.title(class_names[np.argmax(pred[i])], color='green' if ok else 'red', fontsize=8)
    plt.axis('off')
plt.show()
